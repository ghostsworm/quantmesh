# 平仓与订单管理改造 - 技术设计文档

## 一、概述

本文档描述 QuantMesh 做市商系统在平仓方式和订单管理方面的改造设计，主要针对以下需求：

1. **平仓方式选择**：支持市价/限价平仓，状态追踪，超时重试
2. **订单管理**：启动前列出委托，支持修改/删除特定槽位
3. **交易方向选择**：单向做多/单向做空/双向（已支持 LONG/SHORT，需增强为 BOTH）
4. **智能挂单**：渐进式挂单，节省资金占用

#### 3.1.2 核心逻辑设计

**新增模块**: `position/close_manager.go`

```go
package position

import (
    "context"
    "fmt"
    "sync"
    "time"
    "quantmesh/config"
    "quantmesh/logger"
)

// ClosePositionManager 平仓管理器
type ClosePositionManager struct {
    exchange    ExchangeInterface
    botID       string
    symbol      string

    // 平仓记录存储
    records     map[string]*ClosePositionRecord
    recordsMu   sync.RWMutex

    // 超时检查
    ctx         context.Context
    cancel      context.CancelFunc
    wg          sync.WaitGroup
}

// NewClosePositionManager 创建平仓管理器
func NewClosePositionManager(exchange ExchangeInterface, botID, symbol string) *ClosePositionManager {
    ctx, cancel := context.WithCancel(context.Background())
    return &ClosePositionManager{
        exchange: exchange,
        botID:    botID,
        symbol:   symbol,
        records:  make(map[string]*ClosePositionRecord),
        ctx:      ctx,
        cancel:   cancel,
    }
}

// ClosePositions 平仓（支持市价/限价）
func (cpm *ClosePositionManager) ClosePositions(
    ctx context.Context,
    side string,
    quantity float64,
    cfg config.ClosePositionConfig,
) (*ClosePositionRecord, error) {
    record := &ClosePositionRecord{
        RecordID:  generateRecordID(),
        BotID:     cpm.botID,
        Symbol:    cpm.symbol,
        Side:      side,
        TargetQty: quantity,
        Method:    cfg.Method,
        Status:    CloseStatusPending,
        CreatedAt: time.Now(),
        UpdatedAt: time.Now(),
    }

    if cfg.TimeoutSec > 0 {
        record.TimeoutAt = record.CreatedAt.Add(time.Duration(cfg.TimeoutSec) * time.Second)
    }

    // 构建订单请求
    orderReq := &exchange.OrderRequest{
        Symbol:     cpm.symbol,
        Side:       exchange.Side(side),
        Quantity:   quantity,
        ReduceOnly: true,
    }

    if cfg.Method == config.CloseMethodMarket {
        orderReq.Type = exchange.OrderTypeMarket
    } else {
        // 限价单：根据当前价计算限价
        currentPrice, _ := cpm.exchange.GetLatestPrice(ctx, cpm.symbol)
        price := calculateLimitPrice(currentPrice, side, cfg.PriceOffset)
        orderReq.Type = exchange.OrderTypeLimit
        orderReq.Price = price
        orderReq.TimeInForce = exchange.TimeInForceGTC
        record.Price = price
    }

    // 下单
    order, err := cpm.exchange.PlaceOrder(ctx, orderReq)
    if err != nil {
        record.Status = CloseStatusFailed
        record.ErrorMessage = err.Error()
        record.UpdatedAt = time.Now()
        return record, err
    }

    record.OrderID = order.OrderID
    record.Status = CloseStatusPending

    // 保存记录
    cpm.recordsMu.Lock()
    cpm.records[record.RecordID] = record
    cpm.recordsMu.Unlock()

    // 启动超时检查
    if cfg.TimeoutSec > 0 {
        cpm.wg.Add(1)
        go cpm.watchTimeout(record, cfg)
    }

    logger.Info("📤 [平仓] 已下单: %s %s %.4f 方法:%s 订单ID:%d",
        cpm.symbol, side, quantity, cfg.Method, order.OrderID)

    return record, nil
}

// watchTimeout 监控超时
func (cpm *ClosePositionManager) watchTimeout(
    record *ClosePositionRecord,
    cfg config.ClosePositionConfig,
) {
    defer cpm.wg.Done()

    ticker := time.NewTicker(time.Second)
    defer ticker.Stop()

    for {
        select {
        case <-cpm.ctx.Done():
            return
        case <-time.After(time.Until(record.TimeoutAt)):
            // 超时检查
            cpm.recordsMu.Lock()
            if record.Status != CloseStatusPending {
                cpm.recordsMu.Unlock()
                return
            }

            // 检查订单状态
            order, err := cpm.exchange.GetOrder(context.Background(), cpm.symbol, record.OrderID)
            if err == nil && (order.Status == "FILLED" || order.Status == "PARTIALLY_FILLED") {
                record.FilledQty = order.ExecutedQty
                if order.Status == "FILLED" {
                    record.Status = CloseStatusFilled
                }
                record.UpdatedAt = time.Now()
                cpm.recordsMu.Unlock()
                return
            }

            // 确实超时
            if cfg.AutoRetry && record.Method == config.CloseMethodLimit && record.RetryCount < cfg.MaxRetries {
                // 取消原订单
                cpm.exchange.CancelOrder(context.Background(), cpm.symbol, record.OrderID)

                // 重试为市价单
                record.Method = config.CloseMethodMarket
                record.RetryCount++
                record.UpdatedAt = time.Now()

                go func() {
                    _, err := cpm.ClosePositions(context.Background(), record.Side,
                        record.TargetQty-record.FilledQty, config.ClosePositionConfig{
                            Method:     config.CloseMethodMarket,
                            TimeoutSec: cfg.TimeoutSec,
                        })
                    if err != nil {
                        logger.Warn("⚠️ [平仓] 重试失败: %v", err)
                    }
                }()

                cpm.recordsMu.Unlock()
                return
            }

            record.Status = CloseStatusTimeout
            record.UpdatedAt = time.Now()
            cpm.recordsMu.Unlock()

            logger.Warn("⏰ [平仓] 订单超时: RecordID=%s OrderID=%d",
                record.RecordID, record.OrderID)
            return
        }
    }
}

// GetRecord 获取平仓记录
func (cpm *ClosePositionManager) GetRecord(recordID string) (*ClosePositionRecord, bool) {
    cpm.recordsMu.RLock()
    defer cpm.recordsMu.RUnlock()
    record, ok := cpm.records[recordID]
    return record, ok
}

// ListRecords 获取所有平仓记录
func (cpm *ClosePositionManager) ListRecords() []*ClosePositionRecord {
    cpm.recordsMu.RLock()
    defer cpm.recordsMu.RUnlock()

    records := make([]*ClosePositionRecord, 0, len(cpm.records))
    for _, r := range cpm.records {
        records = append(records, r)
    }
    return records
}

// Stop 停止管理器
func (cpm *ClosePositionManager) Stop() {
    cpm.cancel()
    cpm.wg.Wait()
}

// calculateLimitPrice 计算限价
func calculateLimitPrice(currentPrice float64, side string, offsetPercent float64) float64 {
    // offsetPercent: 负数=更激进（卖低价/买高价），正数=更保守
    if side == "SELL" {
        return currentPrice * (1 + offsetPercent/100)
    }
    return currentPrice * (1 - offsetPercent/100)
}

func generateRecordID() string {
    return fmt.Sprintf("close_%d", time.Now().UnixNano())
}
```

```go
package web

import (
    "net/http"
    "quantmesh/config"
    "github.com/gin-gonic/gin"
)

// ClosePositionsV2Request 平仓请求
type ClosePositionsV2Request struct {
    BotID       string                      `json:"bot_id" binding:"required"`
    Method      config.CloseMethod          `json:"method" binding:"required"`      // market/limit
    PriceOffset float64                     `json:"price_offset,omitempty"`         // 限价偏移（%）
    TimeoutSec  int                         `json:"timeout_sec,omitempty"`         // 超时时间（秒）
    AutoRetry   bool                        `json:"auto_retry,omitempty"`          // 是否自动重试
}

// ClosePositionsV2Response 平仓响应
type ClosePositionsV2Response struct {
    Success      bool                        `json:"success"`
    RecordID     string                      `json:"record_id,omitempty"`
    OrderID      int64                       `json:"order_id,omitempty"`
    Status       string                      `json:"status,omitempty"`
    ErrorMessage string                      `json:"error_message,omitempty"`
}

// closePositionsV2 平仓API V2（支持市价/限价）
// POST /api/v2/bots/:id/close-positions
func closePositionsV2(c *gin.Context) {
    botID := c.Param("id")
    var req ClosePositionsV2Request
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }

    // 获取Bot实例
    bot, ok := botManagerProvider.GetBot(botID)
    if !ok {
        c.JSON(http.StatusNotFound, gin.H{"error": "Bot not found"})
        return
    }

    // 调用平仓
    cfg := config.ClosePositionConfig{
        Method:      req.Method,
        PriceOffset: req.PriceOffset,
        TimeoutSec:  req.TimeoutSec,
        AutoRetry:   req.AutoRetry,
        MaxRetries:  3,
    }

    result, err := bot.ClosePositions(cfg)
    if err != nil {
        c.JSON(http.StatusInternalServerError, gin.H{
            "success": false,
            "error":   err.Error(),
        })
        return
    }

    c.JSON(http.StatusOK, ClosePositionsV2Response{
        Success:  true,
        RecordID: result.RecordID,
        OrderID:  result.OrderID,
        Status:   result.Status,
    })
}

// getClosePositionRecords 获取平仓记录
// GET /api/v2/bots/:id/close-records
func getClosePositionRecords(c *gin.Context) {
    botID := c.Param("id")
    bot, ok := botManagerProvider.GetBot(botID)
    if !ok {
        c.JSON(http.StatusNotFound, gin.H{"error": "Bot not found"})
        return
    }

    records := bot.GetCloseRecords()
    c.JSON(http.StatusOK, gin.H{"records": records})
}

// getClosePositionRecord 获取单个平仓记录
// GET /api/v2/close-records/:record_id
func getClosePositionRecord(c *gin.Context) {
    recordID := c.Param("record_id")
    // 通过全局管理器获取记录
    record, ok := closePositionManager.GetRecord(recordID)
    if !ok {
        c.JSON(http.StatusNotFound, gin.H{"error": "Record not found"})
        return
    }

    c.JSON(http.StatusOK, record)
}

// retryClosePosition 重试平仓（手动触发）
// POST /api/v2/close-records/:record_id/retry
func retryClosePosition(c *gin.Context) {
    recordID := c.Param("record_id")

    var req struct {
        Method config.CloseMethod `json:"method" binding:"required"`
    }
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }

    // 取消原订单并重新平仓
    result, err := closePositionManager.Retry(recordID, req.Method)
    if err != nil {
        c.JSON(http.StatusInternalServerError, gin.H{"error": err.Error()})
        return
    }

    c.JSON(http.StatusOK, result)
}
```

```go
// SlotFilterRule 槽位过滤规则
type SlotFilterRule struct {
```

```go
// isSlotEnabled 检查槽位是否启用
func (spm *SuperPositionManager) isSlotEnabled(price float64) bool {
    spm.slotFilterMu.RLock()
    defer spm.slotFilterMu.RUnlock()

    if spm.slotFilter == nil || len(spm.slotFilter.Rules) == 0 {
        return true // 无过滤规则，全部启用
    }

    // 检查每条规则
    for _, rule := range spm.slotFilter.Rules {
        matches := false

        // 检查具体价格列表
        if len(rule.Prices) > 0 {
            for _, p := range rule.Prices {
                if math.Abs(p-price) < 0.000001 { // 浮点数比较
                    matches = true
                    break
                }
            }
        }

        // 检查价格区间
        if !matches && rule.MinPrice > 0 && rule.MaxPrice > 0 {
            if price >= rule.MinPrice && price <= rule.MaxPrice {
                matches = true
            }
        }

        // 根据规则类型返回
        if matches {
            if rule.Type == "exclude" {
                return false // 排除
            }
            if rule.Type == "include" {
                return true // 包含（需要所有规则都是include才返回true）
            }
        }
    }

    // 默认：如果没有include规则匹配，返回true
    // 如果只有exclude规则，返回true（因为没有被排除）
    hasIncludeRule := false
    for _, rule := range spm.slotFilter.Rules {
        if rule.Type == "include" {
            hasIncludeRule = true
            break
        }
    }
    return !hasIncludeRule
}

// SetSlotFilter 设置槽位过滤器
func (spm *SuperPositionManager) SetSlotFilter(filter *config.SlotFilterConfig) {
    spm.slotFilterMu.Lock()
    defer spm.slotFilterMu.Unlock()
    spm.slotFilter = filter

    // 记录日志
    for _, rule := range filter.Rules {
        if rule.Type == "exclude" {
            if len(rule.Prices) > 0 {
                logger.Info("🚫 [槽位过滤] 禁用价格位: %v 原因: %s",
                    rule.Prices, rule.Reason)
            }
            if rule.MinPrice > 0 || rule.MaxPrice > 0 {
                logger.Info("🚫 [槽位过滤] 禁用价格区间: [%.2f, %.2f] 原因: %s",
                    rule.MinPrice, rule.MaxPrice, rule.Reason)
            }
        }
    }

    // 立即触发订单调整，取消被禁用槽位的订单
    spm.cancelFilteredSlotOrders()
}

// cancelFilteredSlotOrders 取消被过滤槽位的订单
func (spm *SuperPositionManager) cancelFilteredSlotOrders() {
    var orderIDsToCancel []int64

    spm.slots.Range(func(key, value interface{}) bool {
        price := key.(float64)
        slot := value.(*InventorySlot)

        if !spm.isSlotEnabled(price) {
            slot.mu.Lock()
            if slot.OrderID != 0 {
                orderIDsToCancel = append(orderIDsToCancel, slot.OrderID)
            }
            slot.mu.Unlock()
        }
        return true
    })

    if len(orderIDsToCancel) > 0 {
        logger.Info("🧹 [槽位过滤] 取消被禁用槽位的订单: %d 个", len(orderIDsToCancel))
        spm.executor.BatchCancelOrders(orderIDsToCancel)
    }
}

// GetSlotFilter 获取当前槽位过滤器
func (spm *SuperPositionManager) GetSlotFilter() *config.SlotFilterConfig {
    spm.slotFilterMu.RLock()
    defer spm.slotFilterMu.RUnlock()
    return spm.slotFilter
}
```

```go
        c.JSON(http.StatusNotFound, gin.H{"error": "Bot not found"})
        return
    }

    var rule config.SlotFilterRule

        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }

        filter = &config.SlotFilterConfig{Rules: []config.SlotFilterRule{}}

    c.JSON(http.StatusOK, gin.H{"ok": true})
}

// removeSlotFilterRule 删除过滤规则
// DELETE /api/bots/:id/slot-filter/rules/:index
func removeSlotFilterRule(c *gin.Context) {
    botID := c.Param("id")
    ruleIndex := c.Param("index")

    idx := parseInt(ruleIndex)
    bot, ok := botManagerProvider.GetBot(botID)
    if !ok {
        c.JSON(http.StatusNotFound, gin.H{"error": "Bot not found"})
        return
    }

    filter := bot.GetSlotFilter()
    if filter == nil || idx >= len(filter.Rules) {
        c.JSON(http.StatusBadRequest, gin.H{"error": "Invalid rule index"})
        return
    }

    filter.Rules = append(filter.Rules[:idx], filter.Rules[idx+1:]...)
    bot.SetSlotFilter(filter)
    c.JSON(http.StatusOK, gin.H{"ok": true})
```

```go
    "math"
    "quantmesh/config"
    "quantmesh/logger"

// SmartOrderManager 智能挂单管理器
type SmartOrderManager struct {
    spm *SuperPositionManager
    cfg *config.SmartOrderConfig
}

// NewSmartOrderManager 创建智能挂单管理器
func NewSmartOrderManager(spm *SuperPositionManager, cfg *config.SmartOrderConfig) *SmartOrderManager {
    return &SmartOrderManager{
        spm: spm,
        cfg: cfg,
    }
}

// CalculateOpenSlots 计算应该挂开仓单的槽位
// 核心思想：只在当前价附近的槽位挂单，避免一次性挂太多
func (som *SmartOrderManager) CalculateOpenSlots(
    currentPrice float64,
    allSlotPrices []float64,
    direction string, // LONG/SHORT
) []float64 {
    if !som.cfg.Enabled {
        // 未启用：返回所有槽位
        return allSlotPrices
    }

    // 1. 按价格排序
    sortedPrices := make([]float64, len(allSlotPrices))
    copy(sortedPrices, allSlotPrices)
    sortFloat64s(sortedPrices, direction)

    // 2. 找到当前价附近的槽位
    var nearbySlots []float64
    maxDistance := som.cfg.OpenOrderDistance * som.spm.config.Trading.PriceInterval

    for _, price := range sortedPrices {
        var distance float64
        if direction == DirectionLong {
            // 做多：关注低于当前价的槽位
            if price < currentPrice {
                distance = currentPrice - price
                if distance <= maxDistance {
                    nearbySlots = append(nearbySlots, price)
                }
            }
        } else {
            // 做空：关注高于当前价的槽位
            if price > currentPrice {
                distance = price - currentPrice
                if distance <= maxDistance {
                    nearbySlots = append(nearbySlots, price)
                }
            }
        }
    }

    // 3. 限制最大数量
    maxOrders := som.cfg.MaxOpenOrders
    if maxOrders <= 0 {
        maxOrders = 3 // 默认最多3个开仓单
    }

    if len(nearbySlots) > maxOrders {
        // 取最近的几个槽位
        if direction == DirectionLong {
            // 做多：取最高的几个（最接近当前价）
            nearbySlots = nearbySlots[len(nearbySlots)-maxOrders:]
        } else {
            // 做空：取最低的几个（最接近当前价）
            nearbySlots = nearbySlots[:maxOrders]
        }
    }

    logger.Debug("🧠 [智能挂单] 当前价:%.2f 方式:%s 候选槽位:%d 选中槽位:%d",
        currentPrice, direction, len(allSlotPrices), len(nearbySlots))

    return nearbySlots
}

// CalculateCloseSlots 计算应该挂平仓单的槽位
func (som *SmartOrderManager) CalculateCloseSlots(
    currentPrice float64,
    direction string,
) []float64 {
    // 遍历所有槽位，找出有持仓的
    var closePrices []float64

    som.spm.slots.Range(func(key, value interface{}) bool {
        price := key.(float64)
        slot := value.(*InventorySlot)

        slot.mu.RLock()
        defer slot.mu.RUnlock()

        if slot.PositionStatus == PositionStatusFilled {
            if !som.cfg.OnlyCloseFilledSlots || slot.OrderID == 0 {
                closePrices = append(closePrices, price)
            }
        }
        return true
    })

    // 限制平仓单数量
    maxCloseOrders := som.spm.config.Trading.SellWindowSize
    if maxCloseOrders <= 0 {
        maxCloseOrders = 5
    }

    if len(closePrices) > maxCloseOrders {
        // 按距离当前价的远近排序，优先挂最近的
        sort.Slice(closePrices, func(i, j int) bool {
            distI := math.Abs(closePrices[i] - currentPrice)
            distJ := math.Abs(closePrices[j] - currentPrice)
            return distI < distJ
        })
        closePrices = closePrices[:maxCloseOrders]
    }

    return closePrices
}

// ShouldAddNewSlot 判断是否应该添加新的挂单槽位
// 当价格移动导致原有槽位成交后，判断是否添加新槽位
func (som *SmartOrderManager) ShouldAddNewSlot(
    currentPrice float64,
    direction string,
) bool {
    if !som.cfg.Enabled || !som.cfg.ProgressivePlacement {
        return true // 未启用渐进式，允许添加
    }

    // 统计当前开仓单数量
    var openOrderCount int
    spm := som.spm

    spm.slots.Range(func(key, value interface{}) bool {
        price := key.(float64)
        slot := value.(*InventorySlot)

        slot.mu.RLock()
        defer slot.mu.RUnlock()

        // 检查是否有有效的开仓单
        if slot.OrderStatus == OrderStatusPlaced ||
           slot.OrderStatus == OrderStatusConfirmed {
            // 判断是否是开仓单
            var isOpeningOrder bool
            if direction == DirectionLong && slot.OrderSide == "BUY" {
                isOpeningOrder = true
            } else if direction == DirectionShort && slot.OrderSide == "SELL" {
                isOpeningOrder = true
            }

            if isOpeningOrder {
                openOrderCount++
            }
        }
        return true
    })

    // 如果开仓单数量少于最大值，允许添加
    maxOrders := som.cfg.MaxOpenOrders
    if maxOrders <= 0 {
        maxOrders = 3
    }

    return openOrderCount < maxOrders
}

// sortFloat64s 按方向排序
func sortFloat64s(arr []float64, direction string) {
    if direction == DirectionLong {
        // 做多：升序
        for i := 0; i < len(arr)-1; i++ {
            for j := i + 1; j < len(arr); j++ {
                if arr[i] > arr[j] {
                    arr[i], arr[j] = arr[j], arr[i]
                }
            }
        }
    } else {
        // 做空：降序
        for i := 0; i < len(arr)-1; i++ {
            for j := i + 1; j < len(arr); j++ {
                if arr[i] < arr[j] {
                    arr[i], arr[j] = arr[j], arr[i]
                }
            }
        }
    }
}
```
